import React from 'react';
import { Bookmark, Pencil, MessageSquareText, Layers, Loader2, Sparkles, AlertCircle, Volume2, RefreshCw, Eye } from 'lucide-react';

export const Card = ({
    wordObj,
    isSavedStatus,
    toggleSaveWord,
    showWriting,
    setShowWriting,
    userSentence,
    setUserSentence,
    writingFeedback,
    setWritingFeedback,
    isEvaluating,
    evaluateSentence,
    showAi,
    setShowAi,
    aiData,
    isAiLoading,
    fetchAiData,
    showForms,
    setShowForms,
    isDark,
    t,
    appLang,
    isTranslated,
    setIsTranslated,
    isRevealed,
    setIsRevealed,
    renderClickableText,
    setQuickTx
}) => {
    return (
        <>
            <div className="absolute top-5 right-5 flex gap-2 z-10">
                <button onClick={(e) => { e.stopPropagation(); toggleSaveWord(wordObj); setQuickTx(prev => ({ ...prev, visible: false })); }} className={`p-2 rounded-xl transition-all ${isSavedStatus ? 'bg-rose-500 text-white' : (isDark ? 'bg-slate-700 text-slate-300' : 'bg-slate-100 text-slate-500')}`} title={isSavedStatus ? t.saved : t.save}><Bookmark size={20} fill={isSavedStatus ? "currentColor" : "none"} /></button>
                <button onClick={(e) => { e.stopPropagation(); setShowForms(false); setShowAi(false); setShowWriting(!showWriting); setQuickTx(prev => ({ ...prev, visible: false })); }} className={`p-2 rounded-xl transition-all ${showWriting ? 'bg-emerald-500 text-white' : (isDark ? 'bg-slate-700 text-slate-300' : 'bg-slate-100 text-slate-500')}`}><Pencil size={20} /></button>
                <button onClick={(e) => { e.stopPropagation(); setShowForms(false); setShowWriting(false); if (!showAi) fetchAiData(wordObj.word); setShowAi(!showAi); setQuickTx(prev => ({ ...prev, visible: false })); }} className={`p-2 rounded-xl transition-all ${showAi ? 'bg-indigo-500 text-white' : (isDark ? 'bg-slate-700 text-slate-300' : 'bg-slate-100 text-slate-500')}`}><MessageSquareText size={20} /></button>
                <button onClick={(e) => { e.stopPropagation(); setShowAi(false); setShowWriting(false); setShowForms(!showForms); setQuickTx(prev => ({ ...prev, visible: false })); }} className={`p-2 rounded-xl transition-all ${showForms ? 'bg-amber-500 text-white' : (isDark ? 'bg-slate-700 text-slate-300' : 'bg-slate-100 text-slate-500')}`}><Layers size={20} /></button>
            </div>

            {showWriting ? (
                <div className="flex flex-col h-full animate-fade-in pt-12">
                    <h2 className="text-xl font-black mb-2 text-emerald-600 flex items-center gap-2">Ô£ı´©Å {t.buildSentence}</h2>
                    <p className="text-xs mb-4 opacity-80">"{wordObj.word}" - {t.writeSentence}</p>
                    <textarea value={userSentence} onChange={(e) => setUserSentence(e.target.value)} placeholder={t.typeHere} className={`w-full p-4 rounded-xl border-2 resize-none focus:outline-none focus:ring-2 focus:ring-emerald-500 ${isDark ? 'bg-slate-900 border-slate-700 text-white' : 'bg-slate-50 border-slate-200'}`} rows={4} />
                    {writingFeedback ? (
                        <div className="mt-4 flex-grow overflow-y-auto pr-1 space-y-3 scrollbar-hide">
                            <div className="flex justify-between items-center p-3 rounded-xl bg-emerald-50 dark:bg-emerald-900/20 border border-emerald-200 dark:border-emerald-800">
                                <span className="font-bold text-emerald-700 dark:text-emerald-400">{t.teacherScore}</span><span className="text-2xl font-black text-emerald-600">{writingFeedback.score}/10</span>
                            </div>
                            <div className={`p-3 rounded-xl border ${isDark ? 'bg-slate-900/50 border-slate-700' : 'bg-slate-50 border-slate-200'}`}>
                                <p className="text-[10px] font-black uppercase text-slate-400 mb-1">{t.teacherComment}</p>
                                <p className="text-sm font-medium italic opacity-90">{typeof writingFeedback.feedback === 'object' ? JSON.stringify(writingFeedback.feedback) : (writingFeedback.feedback || writingFeedback.error)}</p>
                            </div>
                            {writingFeedback.correctedSentence && (
                                <div className="p-3 rounded-xl bg-blue-50 dark:bg-blue-900/20 border border-blue-200 dark:border-blue-800">
                                    <p className="text-[10px] font-black uppercase text-blue-500 mb-1">{t.betterVersion}</p>
                                    <p className="text-sm font-bold text-blue-800 dark:text-blue-300">{writingFeedback.correctedSentence}</p>
                                </div>
                            )}
                            <button onClick={(e) => { e.stopPropagation(); setWritingFeedback(null); setUserSentence(""); }} className="w-full py-2 text-xs font-bold text-slate-500 hover:text-emerald-600">{t.rewrite}</button>
                        </div>
                    ) : (
                        <button onClick={(e) => { e.stopPropagation(); evaluateSentence(wordObj.word); }} disabled={!userSentence.trim() || isEvaluating} className="mt-4 w-full py-3 rounded-xl bg-emerald-600 text-white font-bold flex items-center justify-center gap-2 disabled:opacity-50 shadow-lg">
                            {isEvaluating ? <Loader2 className="animate-spin" size={18} /> : <Sparkles size={18} />} {isEvaluating ? t.teacherReading : t.sendToTeacher}
                        </button>
                    )}
                </div>
            ) : showAi ? (
                <div className="flex flex-col h-full animate-fade-in overflow-hidden">
                    <h2 className="text-xl font-black mb-4 mt-2 text-indigo-500">{t.teacherNotes}</h2>
                    {isAiLoading ? (
                        <div className="flex-grow flex flex-col items-center justify-center gap-4">
                            <Loader2 size={40} className="animate-spin text-indigo-500" />
                            <p className="text-sm italic opacity-70">{t.teacherThinking}</p>
                        </div>
                    ) : aiData?.error ? (
                        <div className="flex-grow flex flex-col items-center justify-center text-center p-4">
                            <AlertCircle size={40} className="text-rose-500 mb-2" /><p className="text-sm font-bold text-rose-500">{typeof aiData.error === 'object' ? JSON.stringify(aiData.error) : aiData.error}</p>
                        </div>
                    ) : (
                        <div className="space-y-4 overflow-y-auto pr-2 scrollbar-hide">
                            <div className="p-4 rounded-xl border-2 border-indigo-500/30 bg-indigo-50/50 dark:bg-indigo-900/10">
                                <p className="text-[10px] font-black uppercase tracking-widest text-indigo-600 mb-1">{t.tactic}</p>
                                <p className="text-sm leading-relaxed italic">{typeof aiData?.mnemonic === 'object' ? JSON.stringify(aiData.mnemonic) : (aiData?.mnemonic || '')}</p>
                            </div>
                            <div className="p-4 rounded-xl border-2 border-amber-500/30 bg-amber-50/50 dark:bg-amber-900/10">
                                <p className="text-[10px] font-black uppercase tracking-widest text-amber-600 mb-1">{t.scenario}</p>
                                <p className="text-sm leading-relaxed font-bold">{typeof aiData?.scenario === 'object' ? JSON.stringify(aiData.scenario) : (aiData?.scenario || '')}</p>
                            </div>
                            <div className="space-y-2">
                                <p className="text-[10px] font-black uppercase tracking-widest text-slate-500 pl-1">{t.examples}</p>
                                {Array.isArray(aiData?.sentences) && aiData.sentences.map((s, i) => (
                                    <div key={i} className={`p-3 rounded-xl border ${isDark ? 'bg-slate-900/50 border-slate-700' : 'bg-slate-50 border-slate-200'}`}>
                                        <p className="text-xs font-bold text-slate-400 mb-1">{typeof s.type === 'object' ? '' : s.type}</p>
                                        <p className="text-sm font-bold mb-1">{typeof s.eng === 'object' ? '' : s.eng}</p>
                                        <p className="text-xs italic opacity-70">{typeof s.tr === 'object' ? '' : s.tr}</p>
                                    </div>
                                ))}
                            </div>
                        </div>
                    )}
                </div>
            ) : showForms ? (
                <div className="flex flex-col h-full animate-fade-in pt-12">
                    <h2 className="text-2xl font-black mb-6">{t.wordForms}</h2>
                    <div className="space-y-3 overflow-y-auto pb-4">
                        {wordObj.wordForms?.length > 0 ? wordObj.wordForms.map((wf, idx) => (
                            <div key={idx} className={`p-4 rounded-2xl border ${isDark ? 'bg-slate-900/30 border-slate-700' : 'bg-slate-50/50 border-slate-200'}`}>
                                <div className="text-[10px] uppercase font-bold text-blue-500 mb-1">{appLang === 'tr' ? wf.posTr : wf.pos}</div>
                                <div className="text-xl font-bold">{wf.form}</div>
                            </div>
                        )) : <p className="text-sm italic opacity-50 text-center mt-10">{t.noForms}</p>}
                    </div>
                </div>
            ) : (
                <div className="flex flex-col h-full animate-fade-in relative">
                    <div className="flex items-center gap-3 mb-2">
                        <div className={`inline-flex shrink-0 px-3 py-1 rounded-full text-[9px] font-bold uppercase tracking-widest border ${isDark ? 'bg-blue-900/20 border-blue-800 text-blue-300' : 'bg-blue-50 border-blue-100 text-blue-700'}`}>
                            {appLang === 'tr' ? wordObj.posTr : wordObj.pos}
                        </div>
                        <div className="h-5 flex items-center overflow-hidden">
                            <span className={`text-sm font-black tracking-wide text-amber-500 transition-all duration-300 ${isTranslated ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-full'}`}>
                                {wordObj.trWord}
                            </span>
                        </div>
                    </div>

                    <div className="mb-6 pr-12 shrink-0">
                        <h1 className="text-4xl font-black tracking-tight mb-2">{wordObj.word}</h1>
                        <div className="flex items-center gap-2 opacity-50 text-sm" style={{ fontFamily: '"Arial Unicode MS", "Lucida Sans Unicode", "Segoe UI", sans-serif' }}>
                            <Volume2 size={16} /> {wordObj.phonetic}
                        </div>
                    </div>

                    {!isRevealed ? (
                        <div className="flex-1 flex flex-col items-center justify-center text-center">
                            <button
                                onClick={(e) => { e.stopPropagation(); setIsRevealed(true); }}
                                className={`p-6 rounded-3xl border-2 border-dashed flex flex-col items-center gap-3 transition-all hover:scale-105 ${isDark ? 'border-indigo-500/50 bg-indigo-900/20 text-indigo-400' : 'border-indigo-300 bg-indigo-50 text-indigo-600'}`}
                            >
                                <Eye size={40} className="animate-pulse" />
                                <span className="font-bold text-lg tracking-wide">{t.activeRecallTap}</span>
                            </button>
                            <p className="text-xs opacity-50 mt-4 px-4">{t.activeRecallLocked}</p>
                        </div>
                    ) : (
                        <div className="flex-1 min-h-0 overflow-y-auto scrollbar-hide pr-1 pb-2">
                            <div className="mb-6 animate-fade-in">
                                <p className="text-[10px] font-bold text-blue-400 mb-2">{t.def}</p>
                                <div className="text-lg leading-relaxed font-serif">
                                    {renderClickableText(isTranslated ? wordObj.trDef : wordObj.engDef)}
                                </div>
                            </div>
                            <div className={`mb-4 p-5 rounded-2xl border animate-fade-in ${isDark ? 'bg-slate-900/50 border-slate-700' : 'bg-slate-50 border-slate-200'}`}>
                                <p className="text-[10px] font-bold text-amber-500 mb-2">{t.ex}</p>
                                <p className="text-md italic font-serif leading-relaxed opacity-80">
                                    "{renderClickableText(isTranslated ? wordObj.trExample : wordObj.engExample)}"
                                </p>
                            </div>
                        </div>
                    )}

                    {(!showWriting && !showAi && !showForms && isRevealed) && (
                        <div className="pt-3 flex justify-center shrink-0">
                            <button
                                onClick={(e) => { e.stopPropagation(); setIsTranslated(!isTranslated); setQuickTx(prev => ({ ...prev, visible: false })); }}
                                className={`px-6 py-2.5 rounded-full text-[11px] font-black uppercase tracking-widest flex items-center gap-2 shadow-sm border transition-all active:scale-95 hover:-translate-y-0.5 ${isDark ? 'bg-slate-800 border-slate-700 text-amber-400 hover:bg-slate-700 hover:border-amber-500/50' : 'bg-white border-slate-200 text-blue-600 hover:bg-slate-50 hover:border-blue-300'}`}
                            >
                                <RefreshCw size={14} className={isTranslated ? "rotate-180 transition-transform" : "transition-transform"} />
                                {isTranslated ? t.toEn : t.toTr}
                            </button>
                        </div>
                    )}
                </div>
            )}
        </>
    );
};
